    public function getCatValueSession()
    {
        $this->_coreSession->start();

    }

    public function getProductCollection($categoryId)
    {
        // $categoryId = 'yourcategoryid';
        $category = $this->_categoryFactory->create()->load($categoryId);
        $collection = $this->_productCollectionFactory->create();
        $collection->addAttributeToSelect('*');
        $collection->addCategoryFilter($category);
        $collection->addAttributeToFilter('visibility', \Magento\Catalog\Model\Product\Visibility::VISIBILITY_BOTH);
        $collection->addAttributeToFilter('status',\Magento\Catalog\Model\Product\Attribute\Source\Status::STATUS_ENABLED);
        return $collection;
    }

    public function getCategoryProduct($categoryId)
    {
        $category = $this->_modelCategoryFactory->create()->load($categoryId)->getProductCollection()->addAttributeToSelect('*');
        return $category;
    }

    public function getProductById($id)
	{
		return $this->_productRepository->getById($id);
	}

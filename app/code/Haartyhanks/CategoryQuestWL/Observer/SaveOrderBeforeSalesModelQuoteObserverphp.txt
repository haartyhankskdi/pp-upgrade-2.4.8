namespace Haartyhanks\CategoryQuest\Observer;

use Magento\Framework\DataObject\Copy;
use Magento\Framework\Event\Observer;
use Magento\Framework\Event\ObserverInterface;
use Magento\Quote\Model\Quote;
use Magento\Sales\Model\Order;
use Haartyhanks\CategoryQuest\Helper\CustomCookie;

class SaveOrderBeforeSalesModelQuoteObserver implements ObserverInterface
{

    /**
     * @var Copy
     */
    protected $objectCopyService;
    /**
     * @var CustomCookie
     */
    protected $customCookie;


    /**
     * @param Copy $objectCopyService
     * @param CustomCookie $customCookie
     */
    public function __construct(
        Copy $objectCopyService,
        CustomCookie $customCookie
    ) {
        $this->objectCopyService = $objectCopyService;
        $this->customCookie = $customCookie;
    }

    /**
     * @param Observer $observer
     */
    public function execute(Observer $observer)
    {
        /* @var Order $order */
        $order = $observer->getEvent()->getData('sales_order_item');
        /* @var Quote $quote */
        $quote = $observer->getEvent()->getData('quote_item');
        $quote = ( $quote->getParentItem() ? $quote->getParentItem() : $quote );
        $quote->setQuestionnaireUniqueId($this->customCookie->get());

        $this->objectCopyService->copyFieldsetToTarget('sales_order_item', 'quote_item', $quote, $order);

        return $this;
    }
}

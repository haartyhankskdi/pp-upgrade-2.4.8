<script>
    function hide_para() {
        var h2 = document.querySelectorAll(".accordian-product");
        for (var i = 0; i < h2.length; i++) {
            h2[i].style.display = 'none';
        }
    }

    hide_para();

    function show_para(currElement) {
        //hide_para();
        //console.log(currElement.nextElementSibling);
        // currElement.nextElementSibling.style.display = "block";
        if (currElement.nextElementSibling.style.display == "none") {
            currElement.nextElementSibling.style.display = "block";
        } else {
            currElement.nextElementSibling.style.display = "none";
        }
    }
</script>

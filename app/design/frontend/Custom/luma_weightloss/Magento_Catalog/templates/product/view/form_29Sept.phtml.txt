$filterOptions = [];
$filterOptionStren = [];
foreach ($childProducts as $child) {
    if ($child->getStatus() == 1) {

        $brand = $_helperCatalog->getLabelById('brand', $child->getBrand());

        if (!$brand) {
            $brand = '-';
        }

        if ($child->getPmdname()) {
            $strength = $_helperCatalog->getLabelById('pmdname', $child->getPmdname());
        } else {
            $strength = '-';
        }
        $size = $_helperCatalog->getLabelById('pmdsize', $child->getPmdsize());
        $price = number_format($child->getPrice(), 2);

        $data = array(
            "brand" => $brand,
            "strength" => $strength,
            "pmdsize" => $size,
            "price" => $price
        );
        array_push($attributsArray, $data);
        array_push($filterOptions, $brand);
        array_push($filterOptionStren, $strength);
    }
}

    function myHashChange() {
        require(['jquery'], function($) {
            let x = location.hash;
            let bodyElement = $("body");
            if (x === "#consultation") {
                history.pushState(null, document.title, location.href);
                if (!bodyElement.hasClass("start_consultation")) {
                    $(".start-consultation").trigger('click');
                }
            } else if (x === "#continue_page") {
                history.pushState(null, document.title, location.href);
                if (!bodyElement.hasClass("swatches_body")) {
                    $(".go-to-start-consultation").trigger('click');
                }
            } else if (x === "" || x === "#") {
                history.pushState(null, document.title, location.href);
                $(".before-start-consultation").trigger('click');
            }
        });
    }
